atmTechSupMIB MODULE-IDENTITY
    LAST-UPDATED "202611130000Z"
    ORGANIZATION "Demo ACC (fictional)"
    CONTACT-INFO
        "Demo ACC technical supervision project.
         This is a demonstration MIB under the documentation
         enterprise number 32473 (RFC 5612). It does not describe
         any real organisation or equipment."
    DESCRIPTION
        "Technical supervision of air traffic control equipment:
         equipment health and role, active alarms, and the main
         operational values of radar, ADS-B, surveillance data
         processing, flight data processing, voice communication,
         radio and recording systems."
    REVISION "202611130000Z"
    DESCRIPTION
        "First version, used by the v0.1 simulator and supervisor."
    ::= { enterprises 32473 1 }

AtmTsEquipmentStatus ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION
        "The health of a piece of equipment. This is health only.
         Maintenance is a separate flag (atmTsEqMaintenanceMode)."
    SYNTAX INTEGER {
        operational(1),
        degraded(2),
        failed(3)
    }

AtmTsSeverity ::= TEXTUAL-CONVENTION
    STATUS current
    DESCRIPTION
        "The severity of an alarm, following ITU X.733 and RFC 3877.
         The numbers are identifiers, not an ordering. In order of
         seriousness: critical, major, minor, warning, indeterminate,
         cleared."
    SYNTAX INTEGER {
        cleared(1),
        indeterminate(2),
        critical(3),
        major(4),
        minor(5),
        warning(6)
    }

    STATUS      current

    SYNTAX      INTEGER {
                    none(1),
                    becomeActive(2),
                    restart(3),
                    resetCounters(4)

    DESCRIPTION

         out that command. Reading always returns none. The outcome

         active it succeeds without doing anything."

--
-- Alarms: the table of active alarms, and the objects that carry an
-- alarm identity in notifications
--

atmTsAlarmTable OBJECT-TYPE
    SYNTAX      SEQUENCE OF AtmTsAlarmEntry
    MAX-ACCESS  not-accessible
    STATUS      current
    DESCRIPTION
        "The alarms that are active on this equipment right now. A
         manager reads this table to learn the current alarm state,
         for example after it has missed notifications."
    ::= { atmTsAlarms 1 }

atmTsAlarmEntry OBJECT-TYPE
    SYNTAX      AtmTsAlarmEntry
    MAX-ACCESS  not-accessible
    STATUS      current
    DESCRIPTION
        "One active alarm. An alarm is identified by its code and its
         instance together, so the same alarm can be active for
         several sensors or frequencies at the same time."
    INDEX       { atmTsAlarmCode, atmTsAlarmInstance }
    ::= { atmTsAlarmTable 1 }

AtmTsAlarmEntry ::= SEQUENCE {
    atmTsAlarmCode          Unsigned32,
    atmTsAlarmInstance      Unsigned32,
    atmTsAlarmInstanceName  DisplayString,
    atmTsAlarmSeverity      AtmTsSeverity,
    atmTsAlarmText          DisplayString,
    atmTsAlarmRaisedAt      DateAndTime
}

atmTsAlarmCode OBJECT-TYPE
    SYNTAX      Unsigned32
    MAX-ACCESS  not-accessible
    STATUS      current
    DESCRIPTION
        "The code that identifies the kind of alarm."
    ::= { atmTsAlarmEntry 1 }

atmTsAlarmInstance OBJECT-TYPE
    SYNTAX      Unsigned32
    MAX-ACCESS  not-accessible
    STATUS      current
    DESCRIPTION
        "Which part of the equipment the alarm is about. The value 0
         means the whole equipment. Otherwise it is a sensor index,
         a frequency index or a channel number, depending on the
         kind of equipment."
    ::= { atmTsAlarmEntry 2 }

atmTsAlarmInstanceName OBJECT-TYPE
    SYNTAX      DisplayString (SIZE (0..32))
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION
        "A readable name for the alarm instance, for example RAD-N
         for a sensor or 132.855 for a frequency. It is empty when the
         instance is 0."
    ::= { atmTsAlarmEntry 3 }

atmTsAlarmSeverity OBJECT-TYPE
    SYNTAX      AtmTsSeverity
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION
        "The severity of the alarm."
    ::= { atmTsAlarmEntry 4 }

atmTsAlarmText OBJECT-TYPE
    SYNTAX      DisplayString (SIZE (0..128))
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION
        "A short description of the alarm for the operator."
    ::= { atmTsAlarmEntry 5 }

atmTsAlarmRaisedAt OBJECT-TYPE
    SYNTAX      DateAndTime
    MAX-ACCESS  read-only
    STATUS      current
    DESCRIPTION
        "The time at which the alarm was raised."
    ::= { atmTsAlarmEntry 6 }

atmTsNotifAlarmCode OBJECT-TYPE
    SYNTAX      Unsigned32
    MAX-ACCESS  accessible-for-notify
    STATUS      current
    DESCRIPTION
        "The alarm code, as carried in an alarm notification. It has
         the same meaning as atmTsAlarmCode. It exists because the

    ::= { atmTsAlarms 2 }

atmTsNotifAlarmInstance OBJECT-TYPE
